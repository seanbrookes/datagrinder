<script setup>
import { ref, onMounted, computed, watch, nextTick } from 'vue';
import { useDataStudioStore } from '../../stores/dataStore.js';
import { 
    parseDataset, 
    analyzeDataset, 
    createTableDOM, 
    sortData 
} from '../../dataHelpers.js';

// --- Global State and Refs ---
const store = useDataStudioStore();
const rawInputText = ref('');
const fileInputRef = ref(null);
const dropZoneActive = ref(false);
const tableContainerRef = ref(null); // Ref for the div receiving the innerHTML table

// State for the processed data and sorting
const rawDataset = ref([]);
const processedHeaders = ref([]);
// This is the line the error pointed to, making it a reactive object for robustness.
// Note: It's still fine to use ref({ ... }), but I'm making it a reactive object to see if it resolves the compiler error.
const sortState = ref({ key: null, direction: 'asc' });

// --- Computed Properties ---

// The final data array after sorting (uses pure JS sort function)
const sortedData = computed(() => {
    // Pass the current sort state to the data rendering function
    return sortData(rawDataset.value, sortState.value.key, sortState.value.direction);
});

// --- Data Processing and Rendering ---

/**
 * Main function to process the input text and update global state.
 */
const processInput = () => {
    store.setDatasetFeedback({ success: false, message: 'Processing dataset...', rows: 0, columns: 0, headers: [] });
    
    const parseResult = parseDataset(rawInputText.value);

    if (parseResult.error || !parseResult.data || parseResult.data.length === 0) {
        store.setDatasetFeedback({
            success: false,
            message: `Parsing failed (Type: ${parseResult.type}). Error: ${parseResult.error || 'No data found.'}`,
            rows: 0,
            columns: 0,
            headers: []
        });
        rawDataset.value = [];
        processedHeaders.value = [];
        // Clear old table content
        if (tableContainerRef.value) {
            tableContainerRef.value.innerHTML = '';
        }
        return;
    }

    rawDataset.value = parseResult.data;
    const analysis = analyzeDataset(parseResult.data);
    console.log('|');
    console.log('| DatasetExplorer analysis ', analysis);
    console.log('|');
    processedHeaders.value = analysis.allKeys;

    // Update global state with feedback
    store.setDatasetFeedback({
        success: true,
        message: `Successfully loaded ${parseResult.type} data.`,
        rows: analysis.rows,
        columns: analysis.columns,
        headers: analysis.headers
    });
    
    // Reset sort state when new data is loaded
    sortState.value = { key: null, direction: 'asc' };
    
    // Initial render of the table
    renderTable(rawDataset.value);
};

/**
 * Renders the data table DOM into the container.
 * @param {Array<Object>} data - The data to render.
 */
const renderTable = (data) => {
    if (!tableContainerRef.value) return;

    // Use the current sort state when creating the DOM (for indicators)
    const tableHtml = createTableDOM(data, processedHeaders.value, sortState.value);
    
    // Note: createTableDOM relies on 'window.vueAppInstance.handleSortClick' 
    // being available when the button is clicked.
    tableContainerRef.value.innerHTML = tableHtml;
};

// Watch for changes in sortedData and re-render the table
watch(sortedData, (newData) => {
    renderTable(newData);
});


// --- Drag and Drop Handlers ---

const handleDragEnter = (e) => {
    e.preventDefault();
    dropZoneActive.value = true;
};

const handleDragLeave = (e) => {
    e.preventDefault();
    dropZoneActive.value = false;
};

const handleDrop = (e) => {
    e.preventDefault();
    dropZoneActive.value = false;
    const file = e.dataTransfer.files[0];
    if (file) {
        readFile(file);
    }
};

const handleFileInput = (e) => {
    const file = e.target.files[0];
    if (file) {
        readFile(file);
    }
};

const readFile = (file) => {
    const reader = new FileReader();
    reader.onload = (e) => {
        rawInputText.value = e.target.result;
        processInput();
    };
    reader.readAsText(file);
};


// --- Sorting Logic (Exposed Globally) ---

/**
 * Handles the click event from the dynamic innerHTML table header button.
 * This function MUST be exposed globally for the DOM to call it.
 * @param {Event} event - The DOM click event.
 * @param {string} headerKey - The column key (header name) to sort by.
 */
const handleSortClick = (event, headerKey) => {
    event.stopPropagation();
    
    const currentKey = sortState.value.key;
    const currentDirection = sortState.value.direction;

    if (currentKey === headerKey) {
        // Toggle direction
        sortState.value = { 
            key: headerKey, 
            direction: currentDirection === 'asc' ? 'desc' : 'asc' 
        };
    } else {
        // New key, reset to ascending
        sortState.value = { 
            key: headerKey, 
            direction: 'asc' 
        };
    }
    // Re-rendering is handled by the `watch(sortedData, ...)` effect.
};

// --- Lifecycle Hook ---

onMounted(() => {
    // Expose the necessary function to the window object so the dynamically 
    // created innerHTML buttons can call it.
    window.vueAppInstance = window.vueAppInstance || {};
    window.vueAppInstance.handleSortClick = handleSortClick;
});


// --- Template ---
</script>

<template>
    <div class="dataset-explorer-container">
        <!-- Input Section -->
        <div 
            class="input-section"
            :class="{ 'drop-zone-active': dropZoneActive }"
            @dragenter.prevent="handleDragEnter"
            @dragleave.prevent="handleDragLeave"
            @dragover.prevent
            @drop.prevent="handleDrop"
        >
            <textarea 
                v-model="rawInputText"
                placeholder="Paste your CSV, JSON, or TSV data here..."
                @input="processInput"
                class="data-input-textarea"
            ></textarea>
            
            <div class="file-upload-overlay">
                <p>or drag and drop a file, or click to upload</p>
                <input 
                    type="file" 
                    ref="fileInputRef" 
                    @change="handleFileInput" 
                    class="file-input"
                    accept=".csv,.json,.tsv,.txt"
                />
            </div>
            
            <button @click="fileInputRef.click()" class="upload-button">
                Upload Data File
            </button>
        </div>

        <!-- Feedback and Analysis Section -->
        <div class="analysis-feedback-section">
            <span 
                class="feedback-chip" 
                :class="{ 'success': store.datasetFeedback.success, 'error': !store.datasetFeedback.success }"
            >
                {{ store.datasetFeedback.message }}
            </span>
            <span v-if="store.datasetFeedback.success" class="analysis-chip">
                Rows: {{ store.datasetFeedback.rows }} | Columns: {{ store.datasetFeedback.columns }}
            </span>
            <div class="header-chips">
                <span v-for="header in store.datasetFeedback.headers" :key="header" class="header-chip">
                    {{ header }}
                </span>
            </div>
        </div>

        <!-- Data Table Section (Preview) -->
        <div class="data-table-section">
            <h2 class="section-title">Data Preview ({{ sortedData.length }} rows)</h2>
            <div class="table-container" ref="tableContainerRef">
                <!-- Data table will be rendered here by JavaScript using innerHTML -->
            </div>
        </div>
    </div>
</template>
